# Stem
Conv2d(in_channels=1, out_channels=32,kernel_size=5, padding=2, stride=2)
BatchNorm2d(num_features=32)
ReLU()

# Block 1
Conv2d(in_channels=32, out_channels=32,kernel_size=3, padding=1)
BatchNorm2d(num_features=32)
ReLU()
Conv2d(in_channels=32, out_channels=32,kernel_size=3, padding=1)
BatchNorm2d(num_features=32)
ReLU()
MaxPool2d(kernel_size=2)

# Block 2
Conv2d(in_channels=32, out_channels=64,kernel_size=3, padding=1)
BatchNorm2d(num_features=64)
ReLU()
Conv2d(in_channels=64, out_channels=64,kernel_size=3, padding=1)
BatchNorm2d(num_features=64)
ReLU()
MaxPool2d(kernel_size=2)

# Block 3
Conv2d(in_channels=64, out_channels=128,kernel_size=3, padding=1)
BatchNorm2d(num_features=128)
ReLU()
Conv2d(in_channels=128, out_channels=128,kernel_size=3, padding=1)
BatchNorm2d(num_features=128)
ReLU()
MaxPool2d(kernel_size=2)

# Block 4
Conv2d(in_channels=128, out_channels=256,kernel_size=3, padding=1)
BatchNorm2d(num_features=256)
ReLU()
Conv2d(in_channels=256, out_channels=256,kernel_size=3, padding=1)
BatchNorm2d(num_features=256)
ReLU()
MaxPool2d(kernel_size=2)

# Block 5
Conv2d(in_channels=256, out_channels=256,kernel_size=3, padding=1)
BatchNorm2d(num_features=256)
ReLU()
Conv2d(in_channels=256, out_channels=256,kernel_size=3, padding=1)
BatchNorm2d(num_features=256)
ReLU()
MaxPool2d(kernel_size=2)

# Head
AdaptiveAvgPool2d(output_size=(1,1))
Flatten()
Dropout(p=0.3)
Linear(in_features=256, out_features=1)
